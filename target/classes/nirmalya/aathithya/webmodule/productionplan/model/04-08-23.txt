CREATE DEFINER=`root`@`localhost` PROCEDURE `purchase_report_routines`(in actionType varchar(100),in actionValue text)
BEGIN

DECLARE EXIT handler for sqlexception
  BEGIN
    -- ERROR
  
  GET DIAGNOSTICS CONDITION 1
		@state = RETURNED_SQLSTATE, 
        @message = MESSAGE_TEXT,
        @contraint = CONSTRAINT_NAME, 
        @errorno = MYSQL_ERRNO; 
        ROLLBACK;
        
        IF ((SELECT LOCATE('Duplicate', @message) > 0) or (SELECT LOCATE('Unknown', @message) > 0)) THEN
        
			SELECT SUBSTRING_INDEX(@message,"'",2) into @a;
			SELECT SUBSTRING_INDEX(@a,"'",-1) into @b;
			SELECT REPLACE(@message,@b,'x') into @c;
			
			SELECT errorId INTO @eId FROM tbl_error_handling where sqlErrorCode=@errorno and sqlErrorMessage=@c;
			
			IF (@eId = '') THEN
				INSERT INTO tbl_error_handling (`sqlErrorCode`, `sqlErrorState`, `sqlErrorMessage`, `nlErrorCode`,`nlErrorMessage`,`dbName`, `eStatus`) VALUES (@errorno, @state, @c, @errorno, @c, 'MYSQL' ,true);
				SELECT errorId INTO @eId FROM tbl_error_handling where sqlErrorCode=@errorno and sqlErrorMessage=@c;
			END IF; 
            
        ELSE    
			SELECT errorId INTO @eId FROM tbl_error_handling where sqlErrorCode=@errorno and sqlErrorMessage=@message;
			
			IF (@eId = '') THEN
				INSERT INTO tbl_error_handling (`sqlErrorCode`, `sqlErrorState`, `sqlErrorMessage`, `nlErrorCode`,`nlErrorMessage`,`dbName`, `eStatus`) VALUES (@errorno, @state, @message, @errorno, @message, 'MYSQL' ,true);
				SELECT errorId INTO @eId FROM tbl_error_handling where sqlErrorCode=@errorno and sqlErrorMessage=@message;
			END IF;
        END IF;
        
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT=@eId;
END;

SET @state='', @message='', @contraint='', @errorno='',@eId='', @a='', @b='', @c='';

########################################################
IF(actionType='viewReportData') THEN
		set @vsql = actionValue;
		prepare prastmt from @vsql;
		execute prastmt;
       
	SELECT JSON_OBJECT(
	    'viewReportData', (SELECT CAST(JSON_ARRAYAGG(JSON_OBJECT(
		'poId',TAII_Po_Id,
        'sku' , TAII_Sku_Id,
		'hsn' , TAII_Hsn_Code,
        'itemName' ,(select TPD_Manufacture_Item from tbl_product_dtls where TPD_SKU = TAII_Sku_Id and TOD_Org_Name = @p_org and TOD_Org_Division = @p_orgDiv),
        'item_type',case when TAII_Product_Id = "" or TAII_Product_Id = null or TAII_Product_Id = 'null' then '' else (select TPM_Product_CategoryText from tbl_product_mstr where TPM_Product_Id = TAII_Product_Id and TOD_Org_Name = @p_org and TOD_Org_Division = @p_orgDiv) end,
        'quantity' , TAII_Quantity,
		'unitName' ,(SELECT TSTM_ST_Name FROM tbl_servetype_mstr WHERE TSTM_ServeType = TAII_Uom),
        'gateEntryDate', (select DATE_FORMAT(TAI_Invoice_Date, '%d-%m-%Y') FROM tbl_action_invoice WHERE TAI_InvoiceId = TAII_Inv_Id AND TOD_Org_Name=@p_org and TOD_Org_Division=@p_orgDiv),
        'poDate', case when TAII_Po_Id = "" or TAII_Po_Id = null or TAII_Po_Id = 'WithOutPo' then "" else (select DATE_FORMAT(TPO_CretaedOn, '%d-%m-%Y') from tbl_po where TPO_Id = TAII_Po_Id and TOD_Org_Name = @p_org and TOD_Org_Division = @p_orgDiv) end,
        'vendorId',(select TAI_Vendor FROM tbl_action_invoice WHERE TAI_InvoiceId = TAII_Inv_Id AND TOD_Org_Name=@p_org and TOD_Org_Division=@p_orgDiv ),
        'vendorName',(select TVM_Vendor_Name from tbl_extend_vendor_mstr where  TVM_Vendor = (select TAI_Vendor FROM tbl_action_invoice WHERE TAI_InvoiceId = TAII_Inv_Id AND TOD_Org_Name=@p_org and TOD_Org_Division=@p_orgDiv ) and TOD_Org_Name=@p_org and TOD_Org_Division=@p_orgDiv ),
        'vendorGstNo',(select TVM_Vendor_GST from tbl_extend_vendor_mstr where  TVM_Vendor = (select TAI_Vendor FROM tbl_action_invoice WHERE TAI_InvoiceId = TAII_Inv_Id AND TOD_Org_Name=@p_org and TOD_Org_Division=@p_orgDiv ) and TOD_Org_Name=@p_org and TOD_Org_Division=@p_orgDiv ),
        'billNo', case when TAII_Challan_No = "" or TAII_Challan_No = null or TAII_Challan_No = "null" then "" else TAII_Challan_No end,
        'billDate', case when TAII_Challan_Dt = "" or TAII_Challan_Dt = null or TAII_Challan_Dt = "null" then "" else TAII_Challan_Dt end,
        'transName' ,coalesce((select TAI_TransporterName FROM tbl_action_invoice WHERE TAI_InvoiceId = TAII_Inv_Id AND TOD_Org_Name=@p_org and TOD_Org_Division=@p_orgDiv),""),
        'vehicleNo', coalesce((select TAI_Delivery_Vehicle_No FROM tbl_action_invoice WHERE TAI_InvoiceId = TAII_Inv_Id AND TOD_Org_Name=@p_org and TOD_Org_Division=@p_orgDiv),""),
        'basic_rate' , TAII_Unit_Price,
        'basic_value' , TAII_Line_Total,
        'sgst',TAII_SGST,
        'cgst',TAII_CGST,
        'gst_rate',concat(TAII_Gst_Rate,'%'),
        'igst',TAII_IGST,
        'total_amt',TAII_SItm_TaxableAmt
        
	 )) as JSON ) as json_obj FROM tbl_action_inv_item_dtls  where TOD_Org_Name=@p_org and TOD_Org_Division=@p_orgDiv)
     ) as full_data;
     
END IF;
########################################################
IF(actionType='viewReportFilterData') THEN
		set @vsql = actionValue;
		prepare prastmt from @vsql;
		execute prastmt;
        
        
	SELECT JSON_OBJECT(
	    'viewReportData', (SELECT CAST(JSON_ARRAYAGG(JSON_OBJECT(
		'poId',TGPD_Po_Id,
        'sku' , TGPD_Sku_Id,
		'hsn' , TGPD_Hsn_Code,
        'itemName' , TGPD_Itm_Name,
        'quantity' , TGPD_Quantity,
		'unitName' ,(SELECT TSTM_ST_Name FROM tbl_servetype_mstr WHERE TSTM_ServeType = TGPD_Uom),
        'amount' , TGPD_Line_Total,
        'poDate' ,DATE_FORMAT((SELECT TPO_CretaedOn FROM tbl_po WHERE TPO_Id = TGPD_Po_Id),'%d-%m-%Y'),
        'gateEntryDate' ,DATE_FORMAT((SELECT TGP_Entry_Date FROM tbl_gatepass_entry WHERE TGP_GatePassEntry = TGPD_GatePassEntryId),'%d-%m-%Y'),
        'billNo',TGPD_Challan_No,
        'billDate' , DATE_FORMAT(TGPD_Challan_Date,'%d-%m-%Y'),
	 'vendorName' ,case when TGP_VendorId is null or TGP_VendorId = '' or TGP_VendorId = null or TGP_VendorId = 'null' then TGP_VendorName else (select TVM_Vendor_Name from tbl_extend_vendor_mstr where TVM_Vendor = TGP_VendorId) end,
		 'transName' , (SELECT TGP_Transport_Name FROM tbl_gatepass_entry WHERE TGP_GatePassEntry = TGPD_GatePassEntryId),
        'lrNo' , (SELECT TGP_Lr_No FROM tbl_gatepass_entry WHERE TGP_GatePassEntry = TGPD_GatePassEntryId),
		 'vehicleNo' ,(SELECT TGP_Vehicle_No FROM tbl_gatepass_entry WHERE TGP_GatePassEntry = TGPD_GatePassEntryId),
        'orgName' , tgi.TOD_Org_Name,
        'orgDivision' , tgi.TOD_Org_Division
	 )) as JSON ) as json_obj FROM tbl_gatepass_item_dtls tgi,tbl_gatepass_entry tp  where TGP_GatePassEntry =  TGPD_GatePassEntryId and  tp.TOD_Org_Name=@p_org and tp.TOD_Org_Division=@p_orgDiv and tp.TGP_Entry_Date between @p_fdate and @p_tdate)
     ) as full_data;
     
END IF;
IF(actionType='viewRMPMReportData') THEN
		set @vsql = actionValue;
		prepare prastmt from @vsql;
		execute prastmt;
        
SELECT JSON_OBJECT(
'viewRMPMReportData', (SELECT CAST(JSON_ARRAYAGG(JSON_OBJECT( 
'sku',TPD_SKU,
        'itemName' , TPD_Manufacture_Item,
		'unit' , (SELECT TSTM_ST_Name FROM tbl_servetype_mstr WHERE TSTM_ServeType = TPD_Unit),
        'openingStock',coalesce((select sum(TAII_Receiving_Quantity) from tbl_action_inv_item_dtls where TAII_Sku_Id = TPD_SKU and TOD_Org_Name=@p_org and TOD_Org_Division=@p_orgDiv GROUP BY TAII_Sku_Id),0)
      
         
   )) as JSON ) as json_obj from tbl_product_dtls  where   TOD_Org_Name=@p_org and TOD_Org_Division=@p_orgDiv))
    as full_data;
END IF;

IF(actionType='viewFGReportData') THEN
		set @vsql = actionValue;
		prepare prastmt from @vsql;
		execute prastmt;
        
SELECT JSON_OBJECT(
'viewFGReportData', (SELECT CAST(JSON_ARRAYAGG(JSON_OBJECT( 
'materialCode',TPM_Product_CategoryText,
        'itemName' , TPM_Product_Name,
		'unit' , unitName,
        'sku' , vtpid.TPD_SKU,
         'ofTheDay' , coalesce((select TPID_Quantity from tbl_po_item_dtls where TPID_Sku_Id= vtpid.TPD_SKU and date(TPID_CreatedOn)=(CURDATE())),0),
    'closingStock' , coalesce((select ( select TPID_Quantity from view_purchase_order_items_for_grn where 
    TPID_Product_Id= TPM_Product_Id and  TPID_Sku_Id=vtpid.TPD_SKU ORDER BY date(TPO_CretaedOn) desc LIMIT 1 )
                    -coalesce((select (TAII_Receiving_Quantity + TAII_Received_Quantity) from 
                    view_action_invoice_grn_items  where  TAII_Product_Id= TPM_Product_Id and TAII_Sku_Id=vtpid.TPD_SKU ORDER BY date(TAI_CreatedOn) 
                    desc LIMIT 1),0)),0) , 
	'openingStock' , coalesce((select coalesce((select(select (TAII_Quantity-TAII_Returning_Quantity) from view_invoice_varification_item where 
					  TAII_Product_Id=TPM_Product_Id and TAII_Sku_Id=vtpid.TPD_SKU ORDER BY date(TAI_CreatedOn) desc LIMIT 1) ),0) + 
                      coalesce((select ( select TPID_Quantity from view_purchase_order_items_for_grn where TPID_Product_Id= TPM_Product_Id and 
                      TPID_Sku_Id=vtpid.TPD_SKU ORDER BY date(TPO_CretaedOn) desc LIMIT 1 )-coalesce((select (TAII_Receiving_Quantity + 
                      TAII_Received_Quantity) from 
                      view_action_invoice_grn_items  where  TAII_Product_Id= TPM_Product_Id and 
                      TAII_Sku_Id=vtpid.TPD_SKU ORDER BY date(TAI_CreatedOn) desc LIMIT 1),0)),0)),0) 
   )) as JSON ) as json_obj from view_tbl_product_item_dtls  as vtpid join tbl_packaging_process_dtls as 
   tppd on tppd.TPD_SKU=vtpid.TPD_SKU where vtpid.TOD_Org_Name=@p_org and vtpid.TOD_Org_Division=@p_orgDiv and  vtpid.TPM_Product_Category in ('PCAT000004'))
     ) as full_data;
END IF;

IF(actionType='viewSFGReportData') THEN
		set @vsql = actionValue;
		prepare prastmt from @vsql;
		execute prastmt;
        
SELECT JSON_OBJECT(
'viewSFGReportData', (SELECT CAST(JSON_ARRAYAGG(JSON_OBJECT( 
'materialCode',TPM_Product_CategoryText,
        'itemName' , TPM_Product_Name,
		'unit' , unitName,
        'sku' , TPD_SKU,
        'ofTheDay' , coalesce((select TPID_Quantity from tbl_po_item_dtls where TPID_Sku_Id=TPD_SKU and date(TPID_CreatedOn)=(CURDATE())),0),
    'closingStock' , coalesce((select ( select TPID_Quantity from view_purchase_order_items_for_grn where TPID_Product_Id= TPM_Product_Id and  TPID_Sku_Id=TPD_SKU ORDER BY date(TPO_CretaedOn) desc LIMIT 1 )
                    -coalesce((select (TAII_Receiving_Quantity + TAII_Received_Quantity) from 
                    view_action_invoice_grn_items  where  TAII_Product_Id= TPM_Product_Id and TAII_Sku_Id=TPD_SKU ORDER BY date(TAI_CreatedOn) desc LIMIT 1),0)),0) , 
	'openingStock' , coalesce((select coalesce((select(select (TAII_Quantity-TAII_Returning_Quantity) from view_invoice_varification_item where 
					  TAII_Product_Id=TPM_Product_Id and TAII_Sku_Id=TPD_SKU ORDER BY date(TAI_CreatedOn) desc LIMIT 1) ),0) + coalesce((select ( select TPID_Quantity from view_purchase_order_items_for_grn where TPID_Product_Id= TPM_Product_Id and 
                      TPID_Sku_Id=TPD_SKU ORDER BY date(TPO_CretaedOn) desc LIMIT 1 )-coalesce((select (TAII_Receiving_Quantity + TAII_Received_Quantity) from 
                      view_action_invoice_grn_items  where  TAII_Product_Id= TPM_Product_Id and 
                      TAII_Sku_Id=TPD_SKU ORDER BY date(TAI_CreatedOn) desc LIMIT 1),0)),0)),0) 
   )) as JSON ) as json_obj from view_tbl_product_item_dtls  where TOD_Org_Name=@p_org and TOD_Org_Division=@p_orgDiv and  TPM_Product_Category in ('PCAT000003'))
     ) as full_data;
END IF;
END
